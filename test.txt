import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

import org.junit.jupiter.api.*;
import org.springframework.http.HttpStatus;

class CodeValidatorServiceImplTest {

  private CodeValidatorServiceImpl svc;
  private CodeValidatorServiceImpl spySvc;

  @BeforeEach
  void init() {
    svc    = new CodeValidatorServiceImpl();
    spySvc = spy(svc);
  }

  @Test
  void success() {
    // stub helpers
    doNothing().when(spySvc).writeCodeToFile(anyString());
    doReturn("").when(spySvc).runCheckSyntax();
    doNothing().when(spySvc).deleteValidateFile();

    ResponseAPI resp = spySvc.validateCode("whatever");
    assertEquals("Code is valid", resp.getData());
  }

  @Test
  void syntaxError() {
    doNothing().when(spySvc).writeCodeToFile(anyString());
    doReturn("E:1 undefined\n").when(spySvc).runCheckSyntax();

    DhpAutomationTestException ex = assertThrows(
      DhpAutomationTestException.class,
      () -> spySvc.validateCode("bad")
    );
    assertEquals(HttpStatus.BAD_REQUEST, ex.getStatus());
    assertTrue(ex.getMessage().contains("undefined"));
  }

  @Test
  void writeFails() {
    doThrow(new DhpAutomationTestException("Unable to write code file",
                                          HttpStatus.INTERNAL_SERVER_ERROR))
      .when(spySvc).writeCodeToFile(anyString());

    DhpAutomationTestException ex = assertThrows(
      DhpAutomationTestException.class,
      () -> spySvc.validateCode("x")
    );
    assertEquals("Unable to write code file", ex.getMessage());
  }

  @Test
  void deleteFails() {
    doNothing().when(spySvc).writeCodeToFile(anyString());
    doReturn("").when(spySvc).runCheckSyntax();
    doThrow(new DhpAutomationTestException("Unable to delete code file",
                                          HttpStatus.INTERNAL_SERVER_ERROR))
      .when(spySvc).deleteValidateFile();

    DhpAutomationTestException ex = assertThrows(
      DhpAutomationTestException.class,
      () -> spySvc.validateCode("x")
    );
    assertEquals("Unable to delete code file", ex.getMessage());
  }

  @Test
  void syntaxCheckException() {
    doNothing().when(spySvc).writeCodeToFile(anyString());
    doThrow(new DhpAutomationTestException("Having issue while validating code",
                                          HttpStatus.INTERNAL_SERVER_ERROR))
      .when(spySvc).runCheckSyntax();

    DhpAutomationTestException ex = assertThrows(
      DhpAutomationTestException.class,
      () -> spySvc.validateCode("x")
    );
    assertEquals("Having issue while validating code", ex.getMessage());
  }
}
