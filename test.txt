import com.azure.core.http.rest.Response;
import com.azure.storage.file.share.ShareClient;
import com.azure.storage.file.share.ShareDirectoryClient;
import com.azure.storage.file.share.ShareFileClient;
import com.azure.storage.file.share.models.ShareFileUploadInfo;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.io.TempDir;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;

import java.io.File;
import java.nio.file.Files;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class ShareClientServiceImplTest {

    @Mock
    private ShareClient shareClient;
    @Mock
    private ShareDirectoryClient rootDirectoryClient;
    @Mock
    private ShareDirectoryClient subDir1;
    @Mock
    private ShareDirectoryClient subDir2;
    @Mock
    private ShareFileClient fileClient;
    @Mock
    private Response<ShareFileUploadInfo> response;

    private ShareClientServiceImpl service;

    @BeforeEach
    void setUp() {
        service = new ShareClientServiceImpl(shareClient);
        when(shareClient.getRootDirectoryClient()).thenReturn(rootDirectoryClient);
    }

    @Test
    void uploadFileToAzureFileShare_success(@TempDir File tempDir) throws Exception {
        File tempFile = new File(tempDir, "test.txt");
        Files.writeString(tempFile.toPath(), "hello");

        String path = "dir1/dir2/file.txt";
        when(shareClient.getFileClient(path)).thenReturn(fileClient);

        when(rootDirectoryClient.getSubdirectoryClient("dir1")).thenReturn(subDir1);
        when(subDir1.exists()).thenReturn(false);
        when(subDir1.getSubdirectoryClient("dir2")).thenReturn(subDir2);
        when(subDir2.exists()).thenReturn(false);

        doNothing().when(subDir1).create();
        doNothing().when(subDir2).create();
        doNothing().when(fileClient).create(tempFile.length());
        when(fileClient.uploadRangeWithResponse(any(), any(), isNull())).thenReturn(response);

        boolean result = service.uploadFileToAzureFileShare(tempFile, path);

        assertTrue(result);
        verify(fileClient, never()).deleteIfExists();
    }

    @Test
    void uploadFileToAzureFileShare_nonExistentFile() {
        File nonexistent = new File("does_not_exist.txt");
        String path = "any/path.txt";
        when(shareClient.getFileClient(path)).thenReturn(fileClient);

        boolean result = service.uploadFileToAzureFileShare(nonexistent, path);

        assertFalse(result);
        verify(fileClient, times(1)).deleteIfExists();
    }

    @Test
    void uploadFileToAzureFileShare_retryFailure(@TempDir File tempDir) throws Exception {
        File tempFile = new File(tempDir, "test2.txt");
        Files.writeString(tempFile.toPath(), "data");

        String path = "dir1/dir2/file2.txt";
        when(shareClient.getFileClient(path)).thenReturn(fileClient);

        when(rootDirectoryClient.getSubdirectoryClient("dir1")).thenReturn(subDir1);
        when(subDir1.exists()).thenReturn(false);
        when(subDir1.getSubdirectoryClient("dir2")).thenReturn(subDir2);
        when(subDir2.exists()).thenReturn(false);

        doNothing().when(subDir1).create();
        doNothing().when(subDir2).create();
        doNothing().when(fileClient).create(tempFile.length());
        when(fileClient.uploadRangeWithResponse(any(), any(), isNull())).thenReturn(null);

        boolean result = service.uploadFileToAzureFileShare(tempFile, path);

        assertFalse(result);
        verify(fileClient, times(1)).deleteIfExists();
    }
}
