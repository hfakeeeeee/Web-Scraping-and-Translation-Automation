    @Test
    void runTest_normalTest_whenRunDetailIsNull_reportsFailure() {
        when(defaultConfig.getHostName()).thenReturn("host-agent");
        when(defaultConfig.getTestPlatformStrategy()).thenReturn(testPlatformStrategy);
        when(updateRequestService.getRequest("u1", RunDetailDTO.class)).thenReturn(null);

        service.runTest("normal_test/u1");

        ArgumentCaptor<UpdateRunDetailDTO> captor = ArgumentCaptor.forClass(UpdateRunDetailDTO.class);
        verify(updateRequestService).updateRequest(eq("u1"), captor.capture());
        UpdateRunDetailDTO dto = captor.getValue();
        assertEquals(Status.FAILED, dto.getStatus());
        assertTrue(dto.getErrorMessage().contains("Unable to get run detail"));
        verify(terminalProcess).cleanUpTest();
        verify(terminalProcess).closeProcess();
        verify(logsService).end();
    }

    @Test
    void runTest_normalTest_whenCanceled_doesNothingExceptCleanup() {
        when(defaultConfig.getHostName()).thenReturn("host-agent");
        when(defaultConfig.getTestPlatformStrategy()).thenReturn(testPlatformStrategy);
        RunDetailDTO rd = new RunDetailDTO();
        rd.setStatus(Status.CANCELED);
        when(updateRequestService.getRequest("u2", RunDetailDTO.class)).thenReturn(rd);

        service.runTest("normal_test/u2");

        verify(updateRequestService, never()).updateRequest(eq("u2"), any());
        verify(terminalProcess).cleanUpTest();
        verify(terminalProcess).closeProcess();
        verify(logsService).end();
    }

    @Test
    void runTest_normalTest_successfulFlow_reportsInProgressAndFinal() {
        when(defaultConfig.getHostName()).thenReturn("host-agent");
        when(defaultConfig.getTestPlatformStrategy()).thenReturn(testPlatformStrategy);
        RunDetailDTO rd = new RunDetailDTO();
        rd.setLanguage(Language.JAVA);
        rd.setStatus(null);
        rd.setVariables(new HashMap<>());
        rd.setSetupCommand(null);
        when(updateRequestService.getRequest("u3", RunDetailDTO.class)).thenReturn(rd);
        when(testPlatformStrategy.getWorkingDirectory(rd)).thenReturn(new File("."));
        when(testPlatformStrategy.buildRunTestCommand(rd)).thenReturn("run-cmd");
        ExecuteCommandDTO exec = mock(ExecuteCommandDTO.class);
        when(exec.getExitCode()).thenReturn(0);
        when(exec.getErrorMessage()).thenReturn(null);
        when(exec.getMessage()).thenReturn("OK");
        when(terminalProcess.executeTest(eq(Language.JAVA), any(File.class), eq("run-cmd"), anyMap()))
            .thenReturn(exec);

        service.runTest("normal_test/u3");

        ArgumentCaptor<UpdateRunDetailDTO> captor = ArgumentCaptor.forClass(UpdateRunDetailDTO.class);
        verify(updateRequestService, times(2)).updateRequest(eq("u3"), captor.capture());
        List<UpdateRunDetailDTO> calls = captor.getAllValues();
        assertEquals(Status.IN_PROGRESS, calls.get(0).getStatus());
        assertEquals("host-agent", calls.get(0).getAgent());
        assertEquals("host-agent", calls.get(1).getAgent());
        verify(collectService).collectTestReportAndRecords(any(UpdateRunDetailDTO.class), eq(rd));
        verify(terminalProcess).cleanUpTest();
        verify(terminalProcess).closeProcess();
        verify(logsService).end();
    }

    @Test
    void copyConfigFiles_success_and_failure() throws Exception {
        String base = Files.createTempDirectory("cfg").toString();
        VersionControlDTO vc = new VersionControlDTO();
        ConfigFileDTO c = new ConfigFileDTO();
        c.setDestination("");
        c.setFileName("f.txt");
        c.setFileContent(Base64.getEncoder().encodeToString("hello".getBytes()));
        vc.setConfigFiles(List.of(c));
        try (MockedStatic<CryptoUtil> cm = mockStatic(CryptoUtil.class)) {
            cm.when(() -> CryptoUtil.decrypt(anyString())).thenReturn("decrypted");
            ReflectionTestUtils.invokeMethod(service.getClass(), "copyConfigFiles", vc, base);
            String content = Files.readString(new File(base, "f.txt").toPath());
            assertEquals("decrypted", content);
        }
        VersionControlDTO vc2 = new VersionControlDTO();
        ConfigFileDTO c2 = new ConfigFileDTO();
        c2.setDestination("");
        c2.setFileName("g.txt");
        c2.setFileContent("bad!");
        vc2.setConfigFiles(List.of(c2));
        try (MockedStatic<CryptoUtil> cm = mockStatic(CryptoUtil.class)) {
            cm.when(() -> CryptoUtil.decrypt(anyString())).thenThrow(new IOException("fail"));
            assertThrows(DhpAutomationTestException.class,
                () -> ReflectionTestUtils.invokeMethod(service.getClass(), "copyConfigFiles", vc2, base));
        }
    }

    @Test
    void getXPath_success_and_failure() {
        when(defaultConfig.getTestPlatformStrategy()).thenReturn(testPlatformStrategy);
        ExecuteCommandDTO exec = mock(ExecuteCommandDTO.class);
        when(exec.getExitCode()).thenReturn(0);
        when(exec.getErrorMessage()).thenReturn(null);
        when(terminalProcess.executeTest(eq(Language.PYTHON), any(File.class), anyString(), anyMap()))
            .thenReturn(exec);
        byte[] data = "xyz".getBytes(StandardCharsets.UTF_8);
        try (MockedStatic<FileProcessUtils> fm = mockStatic(FileProcessUtils.class)) {
            fm.when(() -> FileProcessUtils.readFile(anyString())).thenReturn(data);
            byte[] out = (byte[]) ReflectionTestUtils.invokeMethod(service, "getXPath", "u");
            assertArrayEquals(data, out);
        }
        ExecuteCommandDTO exec2 = mock(ExecuteCommandDTO.class);
        when(exec2.getExitCode()).thenReturn(1);
        when(exec2.getErrorMessage()).thenReturn("err");
        when(terminalProcess.executeTest(eq(Language.PYTHON), any(File.class), anyString(), anyMap()))
            .thenReturn(exec2);
        assertThrows(DhpAutomationTestException.class,
            () -> ReflectionTestUtils.invokeMethod(service, "getXPath", "u"));
    }
