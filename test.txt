import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.ArgumentMatchers.*;
import static org.mockito.Mockito.*;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.concurrent.TimeUnit;

import org.junit.jupiter.api.Test;
import org.mockito.MockedConstruction;
import org.mockito.MockedStatic;
import org.mockito.junit.jupiter.MockitoExtension;
import org.junit.jupiter.api.extension.ExtendWith;
import org.springframework.http.HttpStatus;

@ExtendWith(MockitoExtension.class)
class CodeValidatorServiceImplTest {

    private final CodeValidatorServiceImpl service = new CodeValidatorServiceImpl();

    @Test
    void validateCode_success() throws Exception {
        // Arrange: static Files + ProcessBuilder mocking
        try (MockedStatic<Files> files = mockStatic(Files.class);
             MockedConstruction<ProcessBuilder> pbc = mockConstruction(ProcessBuilder.class,
                 (pbMock, ctx) -> {
                     // chain directory, redirectErrorStream, command
                     when(pbMock.directory(any())).thenReturn(pbMock);
                     when(pbMock.redirectErrorStream(true)).thenReturn(pbMock);
                     when(pbMock.command(anyString(), anyString(), anyString())).thenReturn(pbMock);

                     // fake process
                     Process proc = mock(Process.class);
                     when(pbMock.start()).thenReturn(proc);
                     when(proc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
                     // empty output → valid code
                     when(proc.getInputStream())
                         .thenReturn(new ByteArrayInputStream("".getBytes()));
                 })) {

            // write & delete succeed
            files.when(() -> Files.write(any(), any())).thenReturn(Paths.get("dummy"));
            files.when(() -> Files.deleteIfExists(any())).thenReturn(true);

            // Act
            ResponseAPI resp = service.validateCode("print('hi')");

            // Assert
            assertNotNull(resp);
            assertEquals("Code is valid", resp.getData());
        }
    }

    @Test
    void validateCode_syntaxError() throws Exception {
        try (MockedStatic<Files> files = mockStatic(Files.class);
             MockedConstruction<ProcessBuilder> pbc = mockConstruction(ProcessBuilder.class,
                 (pbMock, ctx) -> {
                     when(pbMock.directory(any())).thenReturn(pbMock);
                     when(pbMock.redirectErrorStream(true)).thenReturn(pbMock);
                     when(pbMock.command(anyString(), anyString(), anyString())).thenReturn(pbMock);

                     Process proc = mock(Process.class);
                     when(pbMock.start()).thenReturn(proc);
                     when(proc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
                     // non-empty output → simulate lint error
                     when(proc.getInputStream())
                         .thenReturn(new ByteArrayInputStream("Syntax error".getBytes()));
                 })) {

            files.when(() -> Files.write(any(), any())).thenReturn(Paths.get("dummy"));

            // Act & Assert
            DhpAutomationTestException ex = assertThrows(
                DhpAutomationTestException.class,
                () -> service.validateCode("bad code")
            );
            assertEquals(HttpStatus.BAD_REQUEST, ex.getStatus());
            assertTrue(ex.getMessage().contains("Syntax error"));
        }
    }

    @Test
    void validateCode_writeIOException() {
        try (MockedStatic<Files> files = mockStatic(Files.class)) {
            // write fails immediately
            files.when(() -> Files.write(any(), any()))
                 .thenThrow(new IOException("disk full"));

            DhpAutomationTestException ex = assertThrows(
                DhpAutomationTestException.class,
                () -> service.validateCode("irrelevant")
            );
            assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
            assertTrue(ex.getMessage().contains("Unable to write code file"));
        }
    }

    @Test
    void validateCode_deleteIOException() throws Exception {
        try (MockedStatic<Files> files = mockStatic(Files.class);
             MockedConstruction<ProcessBuilder> pbc = mockConstruction(ProcessBuilder.class,
                 (pbMock, ctx) -> {
                     when(pbMock.directory(any())).thenReturn(pbMock);
                     when(pbMock.redirectErrorStream(true)).thenReturn(pbMock);
                     when(pbMock.command(anyString(), anyString(), anyString())).thenReturn(pbMock);

                     Process proc = mock(Process.class);
                     when(pbMock.start()).thenReturn(proc);
                     when(proc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
                     when(proc.getInputStream())
                         .thenReturn(new ByteArrayInputStream("".getBytes()));
                 })) {

            files.when(() -> Files.write(any(), any())).thenReturn(Paths.get("dummy"));
            // deletion fails after syntax passes
            files.when(() -> Files.deleteIfExists(any()))
                 .thenThrow(new IOException("delete locked"));

            DhpAutomationTestException ex = assertThrows(
                DhpAutomationTestException.class,
                () -> service.validateCode("print('ok')")
            );
            assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
            assertTrue(ex.getMessage().contains("Unable to delete code file"));
        }
    }
}
