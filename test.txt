import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.ArgumentMatchers.*;
import static org.mockito.Mockito.*;

import java.io.IOException;
import java.lang.reflect.Field;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

import com.fasterxml.jackson.databind.JsonNode;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.ArgumentCaptor;
import org.mockito.Mock;
import org.mockito.MockedStatic;
import org.mockito.junit.jupiter.MockitoExtension;

// Make sure you have mockito-inline on the classpath for static mocking
@ExtendWith(MockitoExtension.class)
class UpdateRequestServiceImplTest {

    private static final String BASE_URL = "http://localhost:8080/api";
    private UpdateRequestServiceImpl service;

    @Mock
    private HttpClient mockHttpClient;

    @Mock
    @SuppressWarnings("unchecked")
    private HttpResponse<String> mockResponse;

    @BeforeEach
    void setUp() throws Exception {
        // 1) instantiate with a dummy URL
        service = new UpdateRequestServiceImpl(BASE_URL);

        // 2) inject our mockHttpClient into the private final field
        Field httpClientField = UpdateRequestServiceImpl.class
                .getDeclaredField("httpClient");
        httpClientField.setAccessible(true);
        httpClientField.set(service, mockHttpClient);
    }

    @Test
    void testUpdateRequestSuccess() throws Exception {
        // prepare fake JSON payload
        String json = "{\"data\":{\"result\":\"ok\"}}";
        when(mockResponse.body()).thenReturn(json);
        when(mockHttpClient.send(any(HttpRequest.class), any(HttpResponse.BodyHandler.class)))
            .thenReturn(mockResponse);

        // call
        JsonNode result = service.updateRequest("updatePath", new Object());

        // verify returned payload
        assertNotNull(result);
        assertEquals("ok", result.get("result").asText());

        // capture and inspect the sent HttpRequest
        ArgumentCaptor<HttpRequest> reqCap = ArgumentCaptor.forClass(HttpRequest.class);
        verify(mockHttpClient).send(reqCap.capture(), any());
        HttpRequest sent = reqCap.getValue();
        assertEquals("PUT", sent.method());
        assertEquals(BASE_URL + "/updatePath", sent.uri().toString());
        assertTrue(sent.headers().firstValue("Content-Type").orElse("").contains("application/json"));
    }

    @Test
    void testUpdateRequestException() throws Exception {
        // throw on send()
        when(mockHttpClient.send(any(), any()))
            .thenThrow(new IOException("network error"));

        // call → should catch and return null
        JsonNode result = service.updateRequest("x", new Object());
        assertNull(result);
    }

    @Test
    void testPostRequestSuccess() throws Exception {
        String json = "{\"status\":\"success\"}";
        when(mockResponse.body()).thenReturn(json);
        when(mockHttpClient.send(any(HttpRequest.class), any(HttpResponse.BodyHandler.class)))
            .thenReturn(mockResponse);

        JsonNode node = service.postRequest("submit", new Payload("foo"));

        assertNotNull(node);
        assertEquals("success", node.get("status").asText());

        ArgumentCaptor<HttpRequest> reqCap = ArgumentCaptor.forClass(HttpRequest.class);
        verify(mockHttpClient).send(reqCap.capture(), any());
        HttpRequest sent = reqCap.getValue();
        assertEquals("POST", sent.method());
        assertEquals(BASE_URL + "/submit", sent.uri().toString());
        assertTrue(sent.headers().firstValue("Content-Type").orElse("").contains("application/json"));
    }

    @Test
    void testGetRequestSuccess() throws Exception {
        // stub backend JSON
        String json = "{\"data\":{\"name\":\"Alice\",\"age\":30}}";
        when(mockResponse.body()).thenReturn(json);
        when(mockHttpClient.send(any(HttpRequest.class), any(HttpResponse.BodyHandler.class)))
            .thenReturn(mockResponse);

        // stub the static FileProcessUtils.convertTreeToValue(...)
        Person expected = new Person("Alice",30);
        try (MockedStatic<FileProcessUtils> utilities = mockStatic(FileProcessUtils.class)) {
            utilities.when(() ->
                FileProcessUtils.convertTreeToValue(any(JsonNode.class), eq(Person.class))
            ).thenReturn(expected);

            Person actual = service.getRequest("getPerson", Person.class);
            assertEquals(expected, actual);
        }
    }

    @Test
    void testGetRequestException() throws Exception {
        when(mockHttpClient.send(any(), any()))
            .thenThrow(new InterruptedException("oops"));
        Person result = service.getRequest("p", Person.class);
        assertNull(result);
    }

    // --- helper DTOs used only in tests ---
    static class Payload {
        public String foo;
        public Payload(String foo) { this.foo = foo; }
        public String getFoo() { return foo; }
        public void setFoo(String foo) { this.foo = foo; }
    }

    static class Person {
        private String name;
        private int age;
        public Person() { }
        public Person(String name, int age) {
            this.name = name;
            this.age  = age;
        }
        public String getName() { return name; }
        public void setName(String name) { this.name = name; }
        public int getAge() { return age; }
        public void setAge(int age) { this.age = age; }

        @Override
        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof Person)) return false;
            Person p = (Person) o;
            return age == p.age && name.equals(p.name);
        }
        @Override
        public int hashCode() {
            return name.hashCode() * 31 + age;
        }
    }
}
