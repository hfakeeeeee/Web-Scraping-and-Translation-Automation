import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.ArgumentMatchers.any;
import static org.mockito.Mockito.*;

import java.io.File;
import java.io.IOException;
import java.nio.file.Files;

import org.apache.commons.io.FileUtils;
import org.junit.jupiter.api.Test;
import org.mockito.MockedStatic;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

class FileUtilExceptionPathTest {

    // Bạn có thể dùng một logger test-captor để verify log.error được gọi,
    // nhưng nếu chỉ cần cover line thì chỉ cần gọi method mà không throw.

    @Test
    void cleanupDirectory_whenCleanThrowsIOException_shouldCatchAndNotThrow() throws IOException {
        // Tạo temp dir
        File tempDir = Files.createTempDirectory("test-ex").toFile();
        // Mock static FileUtils.cleanDirectory to throw
        try (MockedStatic<FileUtils> mocked = mockStatic(FileUtils.class)) {
            mocked.when(() -> FileUtils.cleanDirectory(any(File.class)))
                  .thenThrow(new IOException("mocked failure"));

            // Không nên throw ra exception
            assertDoesNotThrow(() -> FileUtil.cleanupDirectory(tempDir.getAbsolutePath()));

            // Verify cleanDirectory có được gọi
            mocked.verify(() -> FileUtils.cleanDirectory(tempDir), times(1));
        } finally {
            tempDir.delete();
        }
    }

    @Test
    void cleanupFile_whenForceDeleteOnExitThrowsIOException_shouldCatchAndNotThrow() throws IOException {
        // Tạo temp file
        File tempFile = Files.createTempFile("test-ex-file", ".txt").toFile();
        try (MockedStatic<FileUtils> mocked = mockStatic(FileUtils.class)) {
            mocked.when(() -> FileUtils.forceDeleteOnExit(any(File.class)))
                  .thenThrow(new IOException("mocked failure"));

            assertDoesNotThrow(() -> FileUtil.cleanupFile(tempFile.getAbsolutePath()));

            mocked.verify(() -> FileUtils.forceDeleteOnExit(tempFile), times(1));
        } finally {
            tempFile.delete();
        }
    }
}
