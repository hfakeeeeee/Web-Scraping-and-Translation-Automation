package com.example.validation;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.ArgumentMatchers.*;
import static org.mockito.Mockito.*;

import java.io.BufferedReader;
import java.io.ByteArrayInputStream;
import java.io.File;
import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Paths;
import java.util.List;
import java.util.concurrent.TimeUnit;

import org.junit.jupiter.api.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.MockedConstruction;
import org.mockito.MockedStatic;
import org.mockito.junit.jupiter.MockitoExtension;
import org.springframework.http.HttpStatus;

import com.example.api.ResponseAPI;
import com.example.exception.DhpAutomationTestException;
import com.example.service.CodeValidatorServiceImpl;

@ExtendWith(MockitoExtension.class)
class CodeValidatorServiceImplTest {

    private CodeValidatorServiceImpl service;
    private MockedStatic<Files> filesMock;
    private MockedConstruction<ProcessBuilder> pbConstruction;

    @BeforeEach
    void setUp() {
        service = new CodeValidatorServiceImpl();
        // mock all Files.write / delete calls
        filesMock = mockStatic(Files.class);
        // intercept any new ProcessBuilder()
        pbConstruction = mockConstruction(ProcessBuilder.class, (builder, ctx) -> {
            // fluent stubbing
            when(builder.directory(any(File.class))).thenReturn(builder);
            when(builder.redirectErrorStream(true)).thenReturn(builder);
            when(builder.command(anyVararg())).thenReturn(builder);
        });
    }

    @AfterEach
    void tearDown() {
        filesMock.close();
        pbConstruction.close();
    }

    @Test
    void testValidateCode_Success() throws Exception {
        String code = "print('ok')";
        // 1) file write + delete succeed
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenReturn(Paths.get("playwright/main.py"));
        filesMock.when(() ->
            Files.deleteIfExists(Paths.get("playwright/main.py"))
        ).thenReturn(true);

        // 2) process returns empty output
        Process mockProc = mock(Process.class);
        // grab the single constructed builder
        ProcessBuilder builder = pbConstruction.constructed().get(0);
        when(builder.start()).thenReturn(mockProc);
        when(mockProc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
        when(mockProc.getInputStream()).thenReturn(new ByteArrayInputStream(new byte[0]));

        ResponseAPI resp = service.validateCode(code);

        assertNotNull(resp);
        assertEquals("Code is valid", resp.getData());
        // no exception means OK
    }

    @Test
    void testValidateCode_SyntaxError_ThrowsBadRequest() throws Exception {
        String code = "broken code";
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenReturn(Paths.get("playwright/main.py"));
        // skip delete stub; it'll never be reached

        Process mockProc = mock(Process.class);
        ProcessBuilder builder = pbConstruction.constructed().get(0);
        when(builder.start()).thenReturn(mockProc);
        when(mockProc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
        // simulate pylint error output
        String errorMsg = "E:1:1 undefined name 'x'\n";
        when(mockProc.getInputStream())
            .thenReturn(new ByteArrayInputStream(errorMsg.getBytes()));

        DhpAutomationTestException ex = assertThrows(
            DhpAutomationTestException.class,
            () -> service.validateCode(code)
        );
        assertEquals(HttpStatus.BAD_REQUEST, ex.getStatus());
        assertTrue(ex.getMessage().contains("undefined name"));
    }

    @Test
    void testValidateCode_WriteIOException() throws Exception {
        String code = "any";
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenThrow(new IOException("disk full"));

        DhpAutomationTestException ex = assertThrows(
            DhpAutomationTestException.class,
            () -> service.validateCode(code)
        );
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
        assertEquals("Unable to write code file", ex.getMessage());
    }

    @Test
    void testValidateCode_DeleteIOException() throws Exception {
        String code = "print()";
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenReturn(Paths.get("playwright/main.py"));
        // first runCheckSyntax returns empty
        Process mockProc = mock(Process.class);
        ProcessBuilder builder = pbConstruction.constructed().get(0);
        when(builder.start()).thenReturn(mockProc);
        when(mockProc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
        when(mockProc.getInputStream())
            .thenReturn(new ByteArrayInputStream(new byte[0]));

        // now delete fails
        filesMock.when(() ->
            Files.deleteIfExists(Paths.get("playwright/main.py"))
        ).thenThrow(new IOException("no perms"));

        DhpAutomationTestException ex = assertThrows(
            DhpAutomationTestException.class,
            () -> service.validateCode(code)
        );
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
        assertEquals("Unable to delete code file", ex.getMessage());
    }

    @Test
    void testValidateCode_ProcessStartIOException() throws Exception {
        String code = "print()";
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenReturn(Paths.get("playwright/main.py"));

        ProcessBuilder builder = pbConstruction.constructed().get(0);
        when(builder.start()).thenThrow(new IOException("can't exec"));

        DhpAutomationTestException ex = assertThrows(
            DhpAutomationTestException.class,
            () -> service.validateCode(code)
        );
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
        assertEquals("Having issue while validating code", ex.getMessage());
    }

    @Test
    void testValidateCode_InterruptedDuringWait() throws Exception {
        String code = "print()";
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenReturn(Paths.get("playwright/main.py"));

        Process mockProc = mock(Process.class);
        ProcessBuilder builder = pbConstruction.constructed().get(0);
        when(builder.start()).thenReturn(mockProc);
        doThrow(new InterruptedException("oops"))
            .when(mockProc).waitFor(1, TimeUnit.MINUTES);

        DhpAutomationTestException ex = assertThrows(
            DhpAutomationTestException.class,
            () -> service.validateCode(code)
        );
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
        assertEquals("Having issue while validating code", ex.getMessage());
    }

    @Test
    void testValidateCode_ReadOutputIOException() throws Exception {
        String code = "print()";
        filesMock.when(() ->
            Files.write(Paths.get("playwright/main.py"), code.getBytes())
        ).thenReturn(Paths.get("playwright/main.py"));

        Process mockProc = mock(Process.class);
        ProcessBuilder builder = pbConstruction.constructed().get(0);
        when(builder.start()).thenReturn(mockProc);
        when(mockProc.waitFor(1, TimeUnit.MINUTES)).thenReturn(true);
        // InputStream that blows up during read
        InputStream badStream = new InputStream() {
            @Override
            public int read() throws IOException {
                throw new IOException("stream broken");
            }
        };
        when(mockProc.getInputStream()).thenReturn(badStream);

        DhpAutomationTestException ex = assertThrows(
            DhpAutomationTestException.class,
            () -> service.validateCode(code)
        );
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, ex.getStatus());
        assertEquals("Unable to read output", ex.getMessage());
    }
}
